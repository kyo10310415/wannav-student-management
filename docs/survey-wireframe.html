📊 データベース設計

新規テーブル設計

1. survey_responses (アンケート回答記録)

CREATE TABLE survey_responses (
  id SERIAL PRIMARY KEY,
  student_id VARCHAR(50) NOT NULL,
  response_month VARCHAR(7) NOT NULL,  -- 'YYYY-MM'
  responded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  UNIQUE(student_id, response_month)
);

CREATE INDEX idx_survey_responses_student ON survey_responses(student_id);
CREATE INDEX idx_survey_responses_month ON survey_responses(response_month);

2. stamp_rally_achievements (スタンプラリー達成記録)

CREATE TABLE stamp_rally_achievements (
  id SERIAL PRIMARY KEY,
  student_id VARCHAR(50) NOT NULL,
  achievement_type VARCHAR(50) NOT NULL,  -- 'initial_80', 'continuous_6', 'catch_up_100', 'reset_6'
  achieved_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  notified_at TIMESTAMP,
  roulette_completed_at TIMESTAMP,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_achievements_student ON stamp_rally_achievements(student_id);

3. roulette_results (ルーレット結果)

CREATE TABLE roulette_results (
  id SERIAL PRIMARY KEY,
  student_id VARCHAR(50) NOT NULL,
  achievement_id INTEGER REFERENCES stamp_rally_achievements(id),
  result VARCHAR(10) NOT NULL,  -- 'win', 'lose'
  probability INTEGER NOT NULL,  -- 50, 100
  grade VARCHAR(5),  -- 'S', 'A', 'B', 'C'
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_roulette_student ON roulette_results(student_id);

⚙️ 特典条件判定ロジック

条件判定フローチャート
📋 基本条件チェック（全員共通）
├─ ステータス == 'アクティブ' ?
├─ 延長審査結果 == '延長' ?
└─ 条件OK → 次へ

🎯 特典条件判定（開始時期による分岐）
├─ 【パターンA】2026/3以前開始
│   ├─ 継続月数 >= 6 ?
│   │   ├─ YES → 回答率 >= 80% ? → ✅ 特典対象 (initial_80)
│   │   └─ NO → パターンCへ
│   └─ 
├─ 【パターンB】2026/4以降開始
│   └─ 直近6ヶ月連続回答 ? → ✅ 特典対象 (continuous_6)
│
├─ 【パターンC】2026/3以前開始 & 継続月数 < 6
│   └─ 2026/4以降の回答率 == 100% ? → ✅ 特典対象 (catch_up_100)
│
└─ 【パターンD】リセット後（全員共通）
    └─ 直近6ヶ月連続回答 ? → ✅ 特典対象 (reset_6)

🎁 特典送信
├─ Discord通知送信
├─ ルーレットURL生成 (student_id埋め込み)
├─ achievement レコード作成
└─ notified_at 更新

🎰 ルーレット確率
├─ リザルト総合 == 'S' → 100% 当たり
└─ それ以外 → 50% / 50%
